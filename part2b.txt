        function handleOptionClick(selectedIndex, optionDiv, q) {
            if (answeredCurrent) return;
            answeredCurrent = true;
            const optionItems = optionsListEl.querySelectorAll('.option-item');
            if (selectedIndex === q.correctIndex) {
                correctCount++;
                optionDiv.classList.add('correct');
            } else {
                optionDiv.classList.add('incorrect');
                optionItems[q.correctIndex].classList.add('correct');
            }
            explanationTextEl.textContent = q.explanation;
            explanationCardEl.style.display = 'flex';
            nextBtnEl.disabled = false;
        }

        nextBtnEl.addEventListener('click', () => {
            currentQuestionIndex++;
            if (currentQuestionIndex < activeQuestions.length) {
                loadQuestion();
            } else {
                showResults();
            }
        });

        function showResults() {
            quizScreenEl.style.display = 'none';
            resultScreenEl.style.display = 'block';
            const total = activeQuestions.length;
            const calculatedScore = correctCount * 20;
            resultSubjectLabelEl.textContent = `${currentSubjectName}: ${correctCount} из ${total}`;
            finalScoreEl.textContent = calculatedScore;
            finalStatsEl.textContent = `Правильных ответов: ${correctCount} из ${total}`;
            try {
                const userRaw = localStorage.getItem('jolBilimUser');
                if (userRaw) {
                    const user = JSON.parse(userRaw);
                    const familyCode = user.familyCode;
                    user.testsCompleted = (user.testsCompleted || 0) + 1;
                    user.score = calculatedScore;
                    if (!user.history) { user.history = []; }
                    const now = new Date();
                    const day = String(now.getDate()).padStart(2, '0');
                    const month = String(now.getMonth() + 1).padStart(2, '0');
                    const dateStr = `${day}.${month}`;
                    const historyItem = { date: dateStr, score: calculatedScore, subject: currentSubjectKey };
                    user.history.unshift(historyItem);
                    localStorage.setItem('jolBilimUser', JSON.stringify(user));
                    if (familyCode) {
                        db.collection('families').doc(familyCode).get()
                            .then((doc) => {
                                let firestoreHistory = [];
                                if (doc.exists && doc.data().history) {
                                    firestoreHistory = doc.data().history;
                                }
                                firestoreHistory.unshift(historyItem);
                                return db.collection('families').doc(familyCode).update({
                                    score: calculatedScore,
                                    testsCompleted: firebase.firestore.FieldValue.increment(1),
                                    history: firestoreHistory
                                });
                            })
                            .catch((error) => { console.error("Error updating Firestore family record: ", error); });
                    }
                }
            } catch(e) { console.error(e); }
        }
    </script>
